dim a1[]={1,# Pin used

    16, # Individual panel width

    16, # Individual panel height

    0} # Horizontal scanning

_w=32      # Width of the matrix 

_u=8       # Player X position

_v=8       # Player Y position

_t=0       # Player tail Y offset

_b=32      # Wall X position

_h=4       # Wall height

_g=6       # Wall gap

_z = 0.5 # speed

_s = 0 # score 

GfxCfg(3, a1, 32, 16, 1)# type 3, 16x16 leds, buffered x1

BtnEn(20,1)

## Run the game!

while 1

    clear(0)

    plyr()

    wall()

    coll()

    Textt(fmt(_s),0x404040, 0, 0)

    show()

    wait(50)

wend


fn plyr()

    if BtnDown(20)

        Beep(4, 2000, 20)

        if _v>0:_v=_v-2:end

            _t=-1

        else

            if _v<15:_v=_v+0.5:end

            _t=1

        end

    Pixel(0x400040, _u, _v)

    Pixel(0x400040, _u+1, _v+_t)

fend  

fn wall()

  _b=_b-_z

  if _b<=0

    _b=31

    _g=4+rnd(2) # new random gap

    _h=2+rnd(6) # new gap height

    if _z < 3 # the game gets harder as you play

        _z=_z+0.5

    end

    Sweep(4, 1000, 5000, 255, 255, 200)

    _s=_s+1

  end

  for i=0 to 3 #the wall is 3 lines thick

    Line(0x004000,_b+i,0,_b+i,_h)

    Line(0x004000,_b+i,_h+_g,_b+i,16)

  next

fend

fn coll()

  if _b != _u && _b != _u-1 :return:end

  if _v<=_h:die():end

  if _v>=_h+_g:die():end

fend 

fn die()

    _s=0

  for i=0 to 20

    _x=(_u-2)+rnd(3)

    _y=(_v-2)+rnd(3)

    Pixel(rnd(64)<<16,_x,_y)

    show()

    Freq(4, ((30-i)*100) + 2000, 10, 0.5)

  next

  _b=31

  _z=0.5

  wait(100)

  Clear(0)

  Clear(0)

  Texts("3",0x404000, 10, 1,2,2)

  Show()

  Sweep(4, 1000, 5000, 255, 255, 500)

  Clear(0)

  Texts("2",0x404000, 10, 1,2,2)

  Show()

  Sweep(4, 1000, 5000, 255, 255, 500)

  Clear(0)

  Texts("1",0x404000, 10,1, 2,2)

  Show()

  Sweep(4, 1000, 5000, 255, 255, 800)

  Clear(0)

fend
