# NeoPixel - Moving sine wave
_h=16:_w=16
_z=3.1415926/6
_i=0
dim a1[]={1,# Pin used
    16, # Individual panel width
    16, # Individual panel height
    0} # Horizontal scanning

GfxCfg(3, a1, 16, 16, 1)# type 3, 16x16 leds, buffered x1
loop()

fn SetPixel(p,r,g,b)
    x = p / _w
    t = p % _w
    
    if (x & 1) = 0
        y = t
    else
        y = (_w-1) - t
    end
    c = r << 16 | g << 8 | b
    Pixel(c, x, y)
fend

fn loop()
    while 1
        clear(0)
        for _x=0 to 15
            _y = 8+sin((_x+_i)*_z)*6
            pxl()
            SetPixel(_p,0,128,0)
        next
        show()
        _i=_i+1
        wait(1)
    wend
fend

# Formula for index into 16x16 NeoPixel Matrix
# p=pxl(_x,_y)
fn pxl()
    _p = _x*_w+(_x&1)*(_w-1)+(1-2*(_x&1))*_y
fend
