fn strlen(b0)
    # return length of the string
    l =  idxof(b0,0)
 
    if l >=0
        return l    
    end
    
    return len(b0)
fend 

fn JsonParser(b1,d,b2,e,b3)
    # b1 = json text
    # d  = depth
    # b2 = key
    # e  = element index
    # b3 = value output, make sure the size is bigger than output value length
 
    l = strlen(b1)
    p = strlen(b2)
 
    a = 0      # current depth
    c = 0      # match counter
 
    for i = 0 to l-1
 
        x = b1[i]
 
        if x = '{' || x = '['
            a = a + 1
        end
 
        if x = '}' || x = ']'
            a = a - 1
        end
 
        if a = d + 1 && x = '"'
 
            j = 0
            t = i + 1
 
            while 1
 
                if t = l
                    break
                end
 
                if j = p
                    break
                end
 
                if b1[t] != b2[j]
                    break
                end
 
                t = t + 1
                j = j + 1
 
            wend
 
            if j = p
 
                if t = l
                    break
                end
 
                if b1[t] = '"'
 
                    if c = e
 
                        while 1
                            if t = l
                                break
                            end
 
                            if b1[t] = ':'
                                break
                            end
 
                            t = t + 1
                        wend
 
                        t = t + 1
 
                        while 1
                            if t = l
                                break
                            end
 
                            if b1[t] != ' '
                                break
                            end
 
                            t = t + 1
                        wend
 
                        n = 0
 
                        while 1
 
                            if t = l
                                break
                            end
 
                            if b1[t] = 0
                                break
                            end
 
                            if b1[t] = ',' || b1[t] = '}' || b1[t] = ']' || b1[t] = 13 || b1[t] = 10
                                break
                            end
 
                            if b1[t] != '"'
                                b3[n] = b1[t]
                                n = n + 1
                            end
 
                            t = t + 1
 
                        wend
 
                        b3[n] = 0
                        return n
                    end
 
                    c = c + 1
                end
            end
        end
 
    next
 
    b3[0] = 0
    return -1
fend

fn ParseCase1()    
    #{
    #  "temperature": 21.5,
    #  "humidity": 48,
    #  "pressure": 1012
    #}
    dim b1[] = "{\"temperature\":21.5,\"humidity\":48,\"pressure\":1012}" # with/without '\n' does not matter
    dim b2[] = "temperature"
    dim b3[20]
    d = 0 # deep 0 - root
    e = 0 # element 0 - return first detected element
    JsonParser(b1,d,b2,e,b3)
    
    # show temperature
    println("Temperature:", b3) # 21.5
fend

    println("Temperature:", b3) # 22.3
fend

    dim b2[] = "temperature"

    # show temperature
    println("Temperature:", b3) # 20.0
    
    d = 1 # deep 1 - element child deep 1
    e = 1 # element 1 - return second detected element

    # show temperature
    println("Temperature:", b3) # 19.8
fend

fn ParseCase4()
    
    #{
    #  "readings": [
    #    { "temperature": 20.1 },
    #    { "temperature": 21.4 }
    # ]
    #}
    
    # with/without '\n' does not matter
    dim b1[] = "{\"readings\":[{\"temperature\":20.1},{\"temperature\":21.4}]}"
    dim b2[] = "temperature"
    dim b3[20]
    
    d = 2 # deep 2 - element child deep 2
    e = 0 # element 0 - return first detected element
    JsonParser(b1,d,b2,e,b3)
    # show temperature
    println("Temperature:", b3) # 20.1
    
    d = 2 # deep 2 - element child deep 2
    e = 1 # element 1 - return second detected element
    JsonParser(b1,d,b2,e,b3)
    # show temperature
    println("Temperature:", b3) # 21.4
    
    # Explain deep is 2 (d = 2)
    # depth=0  {
    # depth=1      "readings":
    # depth=1      [
    # depth=2          {
    # depth=2              "temperature":20.1
    # depth=2          }
    # depth=2          {
    # depth=2              "temperature":21.4
    # depth=2          }
    # depth=1      ]
    # depth=0  }
fend

fn ParseCase5()
    
    #{
    #  "system": {
    #    "name": "station1",
    #    "version": 2
    #  },
    #  "weather": {
    #    "temperature": 16.7,
    #    "wind": {
    #      "speed": 8,
    #      "direction": 120
    #    }
    #  }
    #}
    
    # with/without '\n' does not matter
    dim b1[] = "{\"system\":{\"name\":\"station1\",\"version\":2},\"weather\":{\"temperature\":16.7,\"wind\":{\"speed\":8,\"direction\":120}}}"
    dim b3[20]
    
    dim b2[] = "name"
    JsonParser(b1,1,b2,0,b3)   # d = 1, e = 0, key = "name" => station1
    println("name:", b3) # station1
    
    dim b2[] = "temperature"
    JsonParser(b1,1,b2,0,b3)   # d = 1, e = 0, key = "temperature" =>  16.7
    println("temperature:", b3) # 16.7
    
    dim b2[] = "speed"
    JsonParser(b1,2,b2,0,b3)   # d = 2, e = 0 , key = "speed"=>  8
    println("speed:", b3) # 8
    
fend
