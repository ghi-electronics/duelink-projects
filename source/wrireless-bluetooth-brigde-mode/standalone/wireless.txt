AsIo(1)  # optional, in case user wants to talk to device during initialize ESP 

_m = 0 # _m: 0: Bluetooth, 1: WiFi
_c = 0 # count
_i = 0 # in
_o = 0 # out
_w = 0 # write flag

dim b5[1024] # only used in standalone mode for read fifo
 
fn Xtransfer(m,b) 
# m: transfer dire
#   0:Esp2Dwlk
#   1:Dwlnk2Esp
# b: brigde
#   0: standard, no bridge
#   1: bridge

    if m = 0 #Esp2Dwlk
        if _m = 0 # parser for Bluetooth
        
            ReadLine(1,10) # get line include \r\n
            #u = tickms()
     
            #println("readline take1: ", (tickms()-u))
            # example: +BTDATA:1,abcd\r\n
            # Scan from end -2 (exclude \r\n) to start till see comma
            # only send data after comma
            n = 0
            l = strlen(b0)
           
            x = 8 # should alway 8
     
     
            for i = x to l
                if b0[i] = 0x2C # comma: 
                    i = i + 1 # move to next byte
                    break
                else
                    n = (n * 10) + (b0[i] - 48)   
                end
            next
     
     
            dim b2[n]     
     
            memcpy(b2,0,b0,i,n)
    
        else # parser for wifi
            ReadLine(1,10) # first \r\n
            ReadLine(1,10) # second \r\n
            #example: \r\n+IPD,0,12:Hello World!\r\n  => two \r\n, Bluetooth has one at the end
            x = 7 # should always 7 to start num of data, last ',' "+IPD,0,1:""
     
            l = strlen(b0)
            for i = x to l
                if b0[i] = 0x3A # :
                    i = i + 1 # move to next byte
                    break
                else
                    n = (n * 10) + (b0[i] - 48)   
                end
            next
     
            dim b2[n]     
     
            memcpy(b2,0,b0,i,n)
        end
        
        # Write b2 to downlink
        if b = 1
            DLSerWr(b2) # Write to downlink
        else
            if n > 0
                # copy b2 to fifo
                for i = 0 to n
                    if _c < len(b5)
                        b5[_i] = b2[i]
                        _c = _c + 1
                        _i = _i + 1
                        
                        if _i = len(b5)
                            _i = 0
                        end
                        
                    end
                next
            end
        end
    
    else #Dwlnk2Esp
        if b = 1
            v = ReadLine(2,128) # read from downlink till see \r\n
        else
            v = 1 # support only one byte for now
        end
        
        dim b3[v]
        Memcpy(b3,0,b0,0,v)
            
        if _m = 0 # Bluetooth
            
            dim b4[30]
            sprintf(b4, "AT+BTSPPSEND=0,%d", v)
     
            ATCmd(b4)  
     
            # Wait for '>'
            r = 0
            while (r = 0 )
                if SerB2R() = 0
                    wait(1)
                else
                    if SerRd() = '>'
                        break
                    end
                end
            wend

            # write b3 to ESP
            SerWrs(b3)
     
            # wait for OK
            ATWaitResp(1000)
        else 
     
            dim b4[30]
            sprintf(b4, "AT+CIPSEND=0,%d", v)
     
            ATCmd(b4)
     
            # ESP send ....\r\nOK\r\n\r\n>
            # Wait for '>'
            r = 0
            while (r = 0 )
                if SerB2R() = 0
                    wait(2)
                else
                    if SerRd() = '>'
                        break
                    end
                end
            wend
     
            # write b3 to ESP
            SerWrs(b3)
            ATWaitResp(1000)
        end
        
    end

fend
 

fn Bridge(b)
    if b = 1
        dlmode(2, 0) #DOWNLINK HOST MODE 
    end
 
    wait(10)

    k = 0 # backup state
    s = 0 # current state 
    i = 0 
 
    while 1
        # read the gateway status pin
        if _m = 0
            if dread(_d, 1) = 0
                s = 1
            else
                s = 0
            end
        end    
        if _m = 1 
            if dread(_s, 1) = 0
                s = 1
            else 
                s = 0
            end
        end
 
        if k != s
            k = s
            if s = 1 # connected
                #println("Connected")
                ReadLine(1,100)
            else
                #println("Disconnected")
                ReadLine(1,100)
            end
        end
 
        if s = 1
            # read from ESP, then write to downlink, or Fifo
            if (SerB2R() > 0) 
                wait(1) 
                Xtransfer(0,b)
            end
 
            # read from downlink, or from b0 (user write), the write to ESP
            if DLSer2R() > 0
                wait(1)
                Xtransfer(1,b)
            end
            
            if _w = 1 && b = 0
                Xtransfer(1,0)
                _w = 0
            end
        end
 
        wait(1)
    wend
fend

fn StartBT(b1) 
    # b1: Bluetooth name
    _m = 0
    
    AtCmd("AT+BTSPPINIT=2")
    ATWaitResp(2000)
    
    dim b2[30]
    
    sprintf(b2,"AT+BTNAME=\"%s\"",b1)
    
    AtCmd(b2) 
    ATWaitResp(2000)
    
 
    AtCmd("AT+BTSCANMODE=2")
    ATWaitResp(2000)  
    
    AtCmd("AT+BTSPPSTART") 
    ATWaitResp( 2000)
    
fend 

fn DoTest()
        
    StartBT("bt_name")

fend
