# This call access api.open-meteo.com and return hightemp
# The function take 5-8 seconds to return value.

fn HighTemp() 
    # Use connection 0
    AtCmd("AT+CIPSTART=0,\"TCP\",\"api.open-meteo.com\",80")
    # Wait for OK # up to 4seconds, depend on the TCP connection
    ATWaitResp(4000)
    
    # This string to request weather
    # all
    # dim b1[] = "GET /v1/forecast?latitude=42.1&longitude=-83.3&current_weather=true HTTP/1.1\r\nHost: api.open-meteo.com\r\nConnection: close\r\n\r\n"
    # high temp
    dim b1[] = "GET /v1/forecast?latitude=42.1&longitude=-83.3&daily=temperature_2m_max&forecast_days=1 HTTP/1.1\r\nHost: api.open-meteo.com\r\nConnection: close\r\n\r\n"
    # current temp
    #dim b1[] = "GET /v1/forecast?latitude=42.1&longitude=-83.3&current=temperature_2m HTTP/1.1\r\nHost: api.open-meteo.com\r\nConnection: close\r\n\r\n"

    # Calculate size of the string, assign to b2
    dim b2[50]
    SprintF(b2,"AT+CIPSEND=0,%d",len(b1))
    
    # Let ESP know this size will come
    AtCmd(b2)
    ATWaitResp(1000)
    
    # AT+CIPSEND also send \r\n>
    while (1)
        if SerB2R() = 0
            wait(1)
        else
            if SerRd() = '>'
                break
            end
        end
    wend
    
    # send raw with actual size
    SerWrs(b1)
    
    # Wait for OK, up to 10 second
    ATWaitResp(10000)
    
    # after OK\r\n from ESP, next will be date from TCP
    # Wait for Read data from TCP
    while (SerB2R() < 1)
        wait(1000)
    wend
    
    # Parse socket response
    # Data will send with format: ""\r\n+IPD,0,643:HTTP/1.1 200 OK\r\n...raw socket data\r\n0\r\n\r\n" 
    # 643: actual size
    # 0: socket number
    
    ATWaitResp(10000)
    # Now parse actual size
    x = 9 # Looking for 643, always start from 9 if socket < 10: "\r\n+IPD,0,643:...."
     
    l = idxof(b0,0)
    
    if l < 0
        l = len(b0)
    end
    n = 0
    for i = x to l
        if b0[i] = 0x3A # :
            i = i + 1
            break
        else
            n = (n * 10) + (b0[i] - 48)   
        end 
    next
    
    # update n because 643 include :"HTTP/1.1 200 OK\r\n", we need to exclude it to read the rest
    n = n - (l-i)

    if b0[2] = '+' && n > 0   # make sure "\r\n+IPD...""
        # Clear all b0
        memset(b0,0,len(b0))
        
        # Read raw TCP to b0
        i = 0
        while (i < n)
            if (SerB2R() > 0)
                b0[i] = SerRd() 
                i = i + 1
            end
            wait(1)
        wend
    
        # parse Json
        dim b3[10]
        JsonParser(b0, 1, "temperature_2m_max", 1, b3) 
        
        v = Parse(b3, 1, 4)

        # Discard all if any
        SerDisc() 
    end
    
    AtCmd("AT+CIPCLOSE=0")
    ATWaitResp(1000)
    
    return v
fend
