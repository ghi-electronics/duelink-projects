# In this project:
# Use a distance sensor to scan for and detect objects.
# If an object is detected closer than 15 cm, rotate the fan toward the object and turn the fan on.
# While the object is present within 15 cm, keep the fan on.
# If the object disappears or moves farther than 15 cm, turn the fan off.
# Note: The fan and two servo motors draw much power (higher 500mA) and may require an external power source using a Power Inject module:
# https://www.duelink.com/docs/products/adpwrinj-a
# Device setup:
# servo motor 1: host
# distance: address 1
# servo motor: address 2
# fan: address 3

Alias(dis=_d, angle=_g, offset=_o, distance_add=1, servo_add=2, fan_add=3)

Asio(1)
_d = 0 #distance
_g = 45 #angle
_o = 5 #offset
_a = 0 #address
_f = 15 #distance flag

setup()
loop()

fn setup()
    dlmode(2,0)
    wait(40)
    servost(1,20) 
    select(servo_add)
    cmd("servost(1, 0)")
    select(fan_add)
    cmd("fan(0)")
fend

fn Loop()
    angle = 20
    d = 1
    offset = 5
 
    while 1
        wait(100)
        dis = GetDist()
        servost(1,angle)
        angle = angle + (d*offset)
 
        if (angle > 160)
            d = -1
        end
 
        if (angle < 20)
            d = 1
        end
 
        if (dis < _f && dis > 0)
            dwrite(1,0)
            SetFan(angle,1)
 
            t = tickms()
 
            while (dis < _f) 
                dis = GetDist()
                wait(60)
            wend
 
            SetFan(0,0)
            wait(100)
        end
    wend
fend
 
fn select(a)
    # optimize bus, no need to set address if address is same
    if a != _a
        _a = a
        cmd(fmt("sel(",_a,")"))
    end
fend
 
fn SetFan(a,o)
    if o = 1
        select(servo_add)
        cmd(fmt("servost(1,",a,")"))
        select(fan_add)
        wait(100)
        cmd("fan(95)")
    else
        select(3)
        cmd("fan(0)")
    end
fend
 
fn GetDist()
    select(distance_add)
    d = cmd("Distance()")
    return d
fend
 
fn scan(a)
    servost(1,a)
    return GetDist()
fend
 
Loop()
